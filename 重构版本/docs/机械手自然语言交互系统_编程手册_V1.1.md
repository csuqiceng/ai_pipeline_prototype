```python
def is_wake_word(text: str) -> bool:
    """检查文本是否包含小正唤醒词"""
    for wake in WAKE_WORDS:
        if wake in text:
            return True
    return False

def strip_wake_word(text: str) -> str:
    """去除唤醒词，返回纯指令文本"""
    for wake in WAKE_WORDS:
        if text.startswith(wake):
            return text[len(wake):].strip()
    return text
```

### 2.2 指令参数结构

```python
"""指令参数结构定义 - 严格对齐通讯协议V4.3"""
from dataclasses import dataclass
from typing import Optional


@dataclass
class Func104Params:
    """Func104 - 停止控制"""
    func_id: int = 104
    estop_ctrl: int = 0      # IEEE(2) 急停: 0=不操作, 1=按下, 2=松开
    pause_ctrl: int = 0      # IEEE(4) 暂停: 0=不操作, 1=按下, 2=松开
    cancel_ctrl: int = 0     # IEEE(6) 取消: 0=不操作, 1=按下, 2=松开
    reset_ctrl: int = 0      # IEEE(8) 复位: 0=不操作, 1=复位
    def to_ieee_dict(self) -> dict[int, float]:
        return {0: 104, 2: self.estop_ctrl, 4: self.pause_ctrl,
                6: self.cancel_ctrl, 8: self.reset_ctrl}


@dataclass
class Func106Params:
    """Func106 - 关节轴点动"""
    func_id: int = 106
    axis_no: int = 0         # IEEE(2) 关节轴号0~5(J1~J6)
    pos_val: float = 5.0     # IEEE(4) 目标角度(度)
    spd_pct: float = 50.0    # IEEE(6) 速度百分比(0~150)
```

```python
"""流程库数据结构 - 6步状态机 + 演练/确认/权限"""
import json, os
from dataclasses import dataclass, field, asdict
from typing import Optional
from datetime import datetime
from enum import Enum


class FlowState(Enum):
    IDLE = "idle"; READY = "ready"; RUNNING = "running"
    PAUSED = "paused"; COMPLETED = "completed"; ERROR = "error"


@dataclass
class FlowStep:
    step_id: int; step_type: str = "position"  # position/atomic/delay
    action: str; func_id: int; params: dict = field(default_factory=dict)
    position_name: Optional[str] = None; spd_pct: int = 30; description: str = ""


@dataclass
class FlowEntry:
    name: str; description: str = ""; steps: list[FlowStep] = field(default_factory=list)
    rehearsal_spd: int = 10    # 演练速度
    confirmed: bool = False    # 是否已确认锁定
    created_by: str = ""       # 创建者(engineer/operator)
    loop_count: int = 1; current_step: int = 0; state: str = FlowState.IDLE.value
    created_at: str = ""; updated_at: str = ""


class FlowRegistry:
    VALID_TRANSITIONS = {
        FlowState.IDLE: {FlowState.READY}, FlowState.READY: {FlowState.RUNNING, FlowState.IDLE},
        FlowState.RUNNING: {FlowState.PAUSED, FlowState.COMPLETED, FlowState.ERROR},
        FlowState.PAUSED: {FlowState.RUNNING, FlowState.IDLE},
        FlowState.COMPLETED: {FlowState.IDLE}, FlowState.ERROR: {FlowState.IDLE},
    }
    def __init__(self, filepath: str = "flow_registry.json"):
        self.filepath = filepath; self._flows: dict[str, FlowEntry] = {}; self._load()
    def _load(self):
        if os.path.exists(self.filepath):
            with open(self.filepath, "r", encoding="utf-8") as f:
                for item in json.load(f).get("flows", []):
                    steps = [FlowStep(**s) for s in item.get("steps", [])]
                    e = FlowEntry(**{k: v for k, v in item.items() if k != "steps"})
                    e.steps = steps; self._flows[e.name] = e
    def _save(self):
        data = {"version": "2.1", "updated_at": datetime.now().isoformat(), "flows": [asdict(f) for f in self._flows.values()]}
        with open(self.filepath, "w", encoding="utf-8") as f: json.dump(data, f, ensure_ascii=False, indent=2)
    def add(self, entry: FlowEntry) -> bool:
        if entry.name in self._flows: return False
        entry.created_at = datetime.now().isoformat(); entry.updated_at = entry.created_at
        self._flows[entry.name] = entry; self._save(); return True
    def remove(self, name: str) -> bool:
        if name not in self._flows: return False
        del self._flows[name]; self._save(); return True
    def get(self, name: str) -> Optional[FlowEntry]: return self._flows.get(name)
    def list_all(self) -> list[FlowEntry]: return list(self._flows.values())
    def transition(self, name: str, target: FlowState) -> bool:
        f = self._flows.get(name)
        if f is None: return False
        cur = FlowState(f.state)
        if target not in self.VALID_TRANSITIONS.get(cur, set()): return False
        f.state = target.value; f.updated_at = datetime.now().isoformat(); self._save(); return True
    def advance_step(self, name: str) -> Optional[FlowStep]:
        f = self._flows.get(name)
        if f is None: return None
        f.current_step += 1
        if f.current_step > len(f.steps):
            f.current_step = 0; self.transition(name, FlowState.COMPLETED); return None
        self._save(); return f.steps[f.current_step - 1]
    def reset(self, name: str) -> bool:
        f = self._flows.get(name)
        if f is None: return False
        f.current_step = 0; f.state = FlowState.IDLE.value; self._save(); return True
    def confirm_flow(self, name: str) -> bool:
        """确认锁定流程，锁定后不可修改"""
        f = self._flows.get(name)
        if f is None: return False
        f.confirmed = True; f.updated_at = datetime.now().isoformat(); self._save(); return True
```

```python
import json, time, threading, base64, hmac, hashlib
from typing import Callable, Optional
from dataclasses import dataclass
from urllib.parse import urlencode


@dataclass
class ASRResult:
    text: str; confidence: float; is_final: bool; timestamp: float


class ASREngine:
    def __init__(self, app_id: str, api_key: str, api_secret: str):
        self.app_id = app_id; self.api_key = api_key; self.api_secret = api_secret
        self._is_listening = False; self._callback = None
        self._vad_silence_ms = 1500; self._ws_url = ""; self._thread = None
        self._vad_callback = None  # VAD检测回调

    def set_result_callback(self, cb: Callable[[ASRResult], None]): self._callback = cb

    def set_vad_callback(self, cb: Callable):
        """设置VAD检测回调 - 检测到人声开始后200ms内触发"""
        self._vad_callback = cb

    def initialize(self) -> bool:
        try:
            ts = str(int(time.time()))
            sig = f"host: ws-api.xfyun.cn\ndate: {ts}\nGET /v2/iat HTTP/1.1"
            s = base64.b64encode(hmac.new(self.api_secret.encode(), sig.encode(), hashlib.sha256).digest()).decode()
            a = base64.b64encode(f'api_key="{self.api_key}", algorithm="hmac-sha256", headers="host date request-line", signature="{s}"'.encode()).decode()
            self._ws_url = f"wss://ws-api.xfyun.cn/v2/iat?{urlencode({'authorization': a, 'date': ts, 'host': 'ws-api.xfyun.cn'})}"
            return True
        except Exception as e: print(f"[ASR] init fail: {e}"); return False

    def start_listening(self):
        if self._is_listening: return
        self._is_listening = True
        self._thread = threading.Thread(target=self._listen_loop, daemon=True); self._thread.start()

    def stop_listening(self):
        self._is_listening = False
        if self._thread: self._thread.join(timeout=3.0); self._thread = None

    def _listen_loop(self):
        """语音监听循环 - pyaudio采集 -> websocket发送 -> 回调识别结果"""
        try:
            import websocket, pyaudio
            CHUNK = 4800; audio = pyaudio.PyAudio()
            stream = audio.open(format=pyaudio.paInt16, channels=1, rate=16000, input=True, frames_per_buffer=CHUNK)
            ws_params = json.dumps({
                "common": {"app_id": self.app_id},
                "business": {"language": "zh_cn", "domain": "iat", "accent": "mandarin", "vad_eos": self._vad_silence_ms, "dwa": "wpgs"},
                "data": {"status": 0, "format": "audio/L16;rate=16000", "encoding": "raw", "audio": ""}
            })
            def on_msg(ws, message):
                r = json.loads(message)
                if r.get("code", -1) != 0: return
                text = "".join(cw["w"] for wi in r.get("data",{}).get("result",{}).get("ws",[]) for cw in wi.get("cw",[]))
                is_final = r.get("data",{}).get("status") == 2
                if text and self._callback: self._callback(ASRResult(text, 1.0 if is_final else 0.5, is_final, time.time()))
            def on_open(ws): ws.send(ws_params)
            ws = websocket.WebSocketApp(self._ws_url, on_message=on_msg, on_open=on_open)
            threading.Thread(target=ws.run_forever, daemon=True).start(); time.sleep(0.5)
            fc = 0; vad_fired = False  # VAD是否已触发
            while self._is_listening:
                try:
                    ad = stream.read(CHUNK, exception_on_overflow=False)
                    # 检测到第一帧音频时触发VAD回调
                    if not vad_fired and self._vad_callback:
                        self._vad_callback()
                        vad_fired = True
                    ws.send(json.dumps({"data": {"status": 0 if fc==0 else 1, "format": "audio/L16;rate=16000", "encoding": "raw", "audio": base64.b64encode(ad).decode()}}))
                    fc += 1
                except Exception: break
            try: ws.send(json.dumps({"data": {"status": 2, "format": "audio/L16;rate=16000", "encoding": "raw", "audio": ""}}))
            except: pass
            stream.stop_stream(); stream.close(); audio.terminate(); ws.close()
        except ImportError as e: print(f"[ASR] missing dep: {e}")
        except Exception as e: print(f"[ASR] error: {e}")

    @property
    def is_listening(self) -> bool: return self._is_listening
```

```python
import json, time, threading, base64, hmac, hashlib
from typing import Optional, Callable
from urllib.parse import urlencode


class TTSEngine:
    def __init__(self, app_id: str, api_key: str, api_secret: str):
        self.app_id = app_id; self.api_key = api_key; self.api_secret = api_secret

    def speak(self, text: str):
        if self._is_speaking: return
        self._is_speaking = True
        threading.Thread(target=self._speak_sync, args=(text,), daemon=True).start()

    def _speak_sync(self, text: str):
        try:
            import websocket
            ts = str(int(time.time()))
            sig = f"host: ws-api.xfyun.cn\ndate: {ts}\nGET /v2/tts HTTP/1.1"
            s = base64.b64encode(hmac.new(self.api_secret.encode(), sig.encode(), hashlib.sha256).digest()).decode()
            a = base64.b64encode(f'api_key="{self.api_key}", algorithm="hmac-sha256", headers="host date request-line", signature="{s}"'.encode()).decode()
            url = f"wss://ws-api.xfyun.cn/v2/tts?{urlencode({'authorization': a, 'date': ts, 'host': 'ws-api.xfyun.cn'})}"
            req = json.dumps({
                "common": {"app_id": self.app_id},
                "business": {"aue": "raw", "sfl": 1, "auf": "audio/L16;rate=16000", "vcn": self._voice, "speed": self._speed, "volume": self._volume, "pitch": self._pitch, "tte": "UTF8"},
                "data": {"status": 2, "text": base64.b64encode(text.encode()).decode()}
            })
            chunks = []
            def on_msg(ws, message):
                r = json.loads(message)
                if r.get("code", -1) != 0: return
                ad = r.get("data", {}).get("audio", "")
                if ad: chunks.append(base64.b64decode(ad))
                if r.get("data", {}).get("status") == 2: ws.close()
            def on_open(ws): ws.send(req)
            ws = websocket.WebSocketApp(url, on_message=on_msg, on_open=on_open); ws.run_forever()
            if chunks:
                import pyaudio; audio = pyaudio.PyAudio()
                stream = audio.open(format=pyaudio.paInt16, channels=1, rate=16000, output=True)
                stream.write(b"".join(chunks)); stream.stop_stream(); stream.close(); audio.terminate()
        except Exception as e: print(f"[TTS] error: {e}")
        finally:
            self._is_speaking = False
            if self._complete_cb: self._complete_cb()

    @property
    def is_speaking(self) -> bool: return self._is_speaking
    def stop(self): self._is_speaking = False
```

### 4.1 前缀检查（prefix_check.py）

```python
"""前缀检查 + 小正唤醒词 + 5级语义分层 - V1.1完全重写"""
from dataclasses import dataclass
from enum import Enum
from typing import Optional
import re


class InputType(Enum):
    EMERGENCY = "emergency"   # L5: 应急安全
    COMMAND = "command"       # L3: 常规生产执行
    SYSTEM = "system"         # L4: 系统管理
    PRECHECK = "precheck"     # L2: 工艺研讨
    QUERY = "query"           # L2: 查询
    CHITCHAT = "chitchat"     # L1: 闲聊咨询
    INVALID = "invalid"


@dataclass
class ClassifyResult:
    input_type: InputType
    semantic_level: int       # 1~5
    cleaned_text: str
    matched_prefix: Optional[str] = None
    emergency_code: Optional[str] = None  # 三段式编码


WAKE_WORD = "小正"
EMERGENCY_PATTERN = re.compile(r'急停\s*(\d+)\s*急停|暂停\s*(\d+)\s*暂停|复位\s*(\d+)\s*复位')

# 系统管理关键词(L4)
SYSTEM_KEYWORDS = ["速度", "步长", "工作速度", "演练速度", "切换", "保存", "删除", "设定位置", "新建流程"]
# 工艺研讨关键词(L2)
PRECHECK_KEYWORDS = ["能不能到", "能不能走", "安全吗", "能不能执行", "流程能不能", "可行性"]
# 查询关键词(L2)
QUERY_KEYWORDS = ["查询", "查看", "当前", "现在", "什么", "多少", "哪个", "状态", "位置", "速度", "报警", "通讯"]


def classify_input(text: str) -> ClassifyResult:
    """前缀检查 + 小正唤醒词 + 5级语义分层"""
    t = text.strip()
    if not t:
        return ClassifyResult(InputType.INVALID, 0, "")

    # 1. 先检查三段式应急编码(L5, 无需小正前缀)
    m = EMERGENCY_PATTERN.search(t)
    if m:
        code = m.group(1) or m.group(2) or m.group(3)
        keyword = "急停" if m.group(1) else "暂停" if m.group(2) else "复位"
        return ClassifyResult(InputType.EMERGENCY, 5, t, keyword, code)

    # 2. 检查小正唤醒词
    has_wake = t.startswith((WAKE_WORD, WAKE_WORD + "，", WAKE_WORD + " "))
    if has_wake:
        cmd = t[len(WAKE_WORD):].lstrip("，, ")
    else:
        # 无小正前缀: 检查是否单独提及应急关键词
        for word in ["急停", "暂停", "复位"]:
            if word in t and not EMERGENCY_PATTERN.search(t):
                return ClassifyResult(InputType.CHITCHAT, 1, t, word,
                    None)  # 单独提及不触发
        # 无小正前缀的其他输入视为闲聊
        return ClassifyResult(InputType.CHITCHAT, 1, t)

    # 3. 有小正前缀: 按语义分层
    # L5: 应急指令(小正+急停/暂停/复位)
    for word in ["急停", "暂停", "复位", "解除急停", "取消"]:
        if cmd.startswith(word):
            return ClassifyResult(InputType.EMERGENCY, 5, cmd, word)

    # L4: 系统管理
    for kw in SYSTEM_KEYWORDS:
        if kw in cmd:
            return ClassifyResult(InputType.SYSTEM, 4, cmd, kw)

    # L2: 工艺研讨
    for kw in PRECHECK_KEYWORDS:
        if kw in cmd:
            return ClassifyResult(InputType.PRECHECK, 2, cmd, kw)

    # L2: 查询
    for kw in QUERY_KEYWORDS:
        if cmd.startswith(kw):
            return ClassifyResult(InputType.QUERY, 2, cmd, kw)

    # L3: 常规生产执行(默认)
    return ClassifyResult(InputType.COMMAND, 3, cmd)
```

```python
"""单位归一化 - 步骤4"""
import re
from dataclasses import dataclass
from typing import Optional


@dataclass
class UnitValue:
    value: float; unit: str; original: str


# 标准单位转换
UNIT_CONVERSIONS: dict[str, tuple[str, float]] = {
    "毫米": ("mm", 1.0), "mm": ("mm", 1.0),
    "厘米": ("mm", 10.0), "cm": ("mm", 10.0),
    "米": ("mm", 1000.0), "m": ("mm", 1000.0),
    "丝": ("mm", 0.01), "道": ("mm", 0.01),
    "度": ("deg", 1.0), "弧度": ("deg", 57.2958),
    "毫米每秒": ("mm/s", 1.0), "%": ("%", 1.0),
    "百分之": ("%", 1.0),
}

# 口语单位映射
COLLOQUIAL_UNITS: dict[str, tuple[float, str]] = {
    "一拃": (200.0, "mm"), "一拳": (100.0, "mm"), "一指": (20.0, "mm"),
    "一点点": (1.0, "mm"), "半圈": (180.0, "deg"), "一圈": (360.0, "deg"),
    "龟速": (10.0, "%"), "蜗牛": (10.0, "%"), "慢慢来": (20.0, "%"),
    "正常": (50.0, "%"), "快点": (70.0, "%"), "飞速": (90.0, "%"),
    "全速": (100.0, "%"), "最快": (100.0, "%"), "最慢": (5.0, "%"),
}

# 中文数字
CN_DIGITS = {"零": 0, "一": 1, "二": 2, "三": 3, "四": 4, "五": 5,
             "六": 6, "七": 7, "八": 8, "九": 9, "十": 10,
             "百": 100, "千": 1000, "万": 10000, "两": 2, "半": 0.5}

# 正则模式
NUM_UNIT_PATTERN = re.compile(
    r'(\d+\.?\d*)\s*(毫米|厘米|米|丝|道|度|弧度|毫米每秒|%|百分之)')
CN_NUM_UNIT_PATTERN = re.compile(
    r'([零一二三四五六七八九十百千万两半]+)\s*(毫米|厘米|米|丝|道|度|弧度)')


def _parse_cn_number(text: str) -> float:
    """解析中文数字"""
    if not text: return 0.0
    if text in CN_DIGITS: return float(CN_DIGITS[text])
    result = 0.0; current = 0.0
    for ch in text:
        v = CN_DIGITS.get(ch, 0)
        if v >= 10:
            if current == 0: current = 1.0
            result += current * v; current = 0.0
        else:
            current = v
    result += current
    return result


def normalize_unit(text: str) -> list[UnitValue]:
    """单位归一化: 提取文本中的所有数值+单位"""
    results = []
    # 1. 匹配阿拉伯数字+单位
    for m in NUM_UNIT_PATTERN.finditer(text):
        val = float(m.group(1)); unit_cn = m.group(2)
        std_unit, factor = UNIT_CONVERSIONS.get(unit_cn, (unit_cn, 1.0))
        results.append(UnitValue(val * factor, std_unit, m.group(0)))
    # 2. 匹配中文数字+单位
    for m in CN_NUM_UNIT_PATTERN.finditer(text):
        val = _parse_cn_number(m.group(1)); unit_cn = m.group(2)
        std_unit, factor = UNIT_CONVERSIONS.get(unit_cn, (unit_cn, 1.0))
        results.append(UnitValue(val * factor, std_unit, m.group(0)))
    # 3. 匹配百分比 (如 "50%")
    for m in re.finditer(r'(\d+\.?\d*)\s*%', text):
        results.append(UnitValue(float(m.group(1)), "%", m.group(0)))
    # 4. 匹配口语单位
    for colloquial, (val, unit) in COLLOQUIAL_UNITS.items():
        if colloquial in text:
            results.append(UnitValue(val, unit, colloquial))
    return results
```

```python
DIRECTION_PATTERNS: list[tuple[list[str], str]] = [
    (["往前", "向前", "前进"], "前进"),
    (["往后", "向后", "后退"], "后退"),
    (["往左", "向左", "左移"], "左移"),

    (["顺时针", "顺转"], "Rz正转"),
    (["逆时针", "逆转"], "Rz反转"),
]

# 速度模式匹配
SPEED_PATTERNS: list[tuple[list[str], int]] = [
    (["最快", "全速", "飞速"], 100),
    (["很快", "高速"], 80),
    (["快点", "加速"], 70),
    (["中速", "正常速度"], 50),
    (["慢点", "减速"], 30),
    (["很慢", "慢慢来"], 20),
    (["最慢", "龟速", "蜗牛"], 10),
]

# 位置名称匹配
POSITION_NAMES = ["原点", "起点", "终点", "当前位置", "安全位置", "焊接位置"]


def extract_elements(text: str) -> ExtractedCommand:
    """从归一化文本中提取指令要素"""
    cmd = ExtractedCommand(raw_text=text)
    # 1. 提取动作
    for keywords, action in ACTION_PATTERNS:
        for kw in keywords:
            if kw in text:
                cmd.action = action; break
        if cmd.action: break
    # 2. 提取方向 - 优先检查关节轴方向
    for direction, (axis_no, dir_val) in JOINT_DIRECTION_MAP.items():
        if direction in text:
            cmd.direction = direction
            cmd.action = "关节运动"  # 关节轴点动
            break
    # 再检查笛卡尔方向
    if cmd.direction is None:
        for keywords, direction in DIRECTION_PATTERNS:
            for kw in keywords:
                if kw in text:
                    cmd.direction = direction
                    if cmd.action not in ("移动",):
                        cmd.action = "虚拟轴运动"  # 笛卡尔点动
                    break
            if cmd.direction: break
    # 3. 提取目标位置
    for pos in POSITION_NAMES:
        if pos in text:
            cmd.target_position = pos; break
    # 自定义位置名 (如 "位置A")
    m = re.search(r'(?:示教|记录|保存)\s*(\w+)', text)
    if m and not cmd.target_position:
        cmd.target_position = m.group(1)
    # 4. 提取距离
    m = re.search(r'(\d+\.?\d*)\s*毫米', text)
    if m: cmd.distance_mm = float(m.group(1))
    m = re.search(r'(\d+\.?\d*)\s*厘米', text)
    if m: cmd.distance_mm = float(m.group(1)) * 10
    # 5. 提取角度
    m = re.search(r'(\d+\.?\d*)\s*度', text)
    if m: cmd.angle_deg = float(m.group(1))
    # 6. 提取速度
    for keywords, speed in SPEED_PATTERNS:
        for kw in keywords:
            if kw in text:
                cmd.speed_pct = speed; break
        if cmd.speed_pct is not None: break
    m = re.search(r'(\d+)\s*%', text)
    if m and cmd.speed_pct is None:
        cmd.speed_pct = int(m.group(1))
    # 7. 提取轴号
    m = re.search(r'第(\d+)轴', text)
    if m: cmd.axis_id = int(m.group(1))
    # 8. 提取IO号
    m = re.search(r'IO\s*(\d+)', text, re.IGNORECASE)
    if m: cmd.io_number = int(m.group(1))
    # 9. 提取循环次数
    m = re.search(r'循环\s*(\d+)\s*次', text)
    if m: cmd.loop_count = int(m.group(1))
    return cmd
```

```python
import json

from typing import Optional


@dataclass
class DisambiguateResult:
    action: str; confidence: float; params: dict
    reasoning: str; needs_confirm: bool


SYSTEM_PROMPT = """你是机械手语音指令消歧助手。用户输入自然语言指令，你需要:
1. 判断用户意图的动作(移动/关节运动/虚拟轴运动/回零/急停/停止/暂停/继续/示教/标定/打开/关闭/执行/复位)
2. 提取参数(方向/位置/距离/角度/速度/轴号/IO号)
3. 判断是否需要确认
输出JSON格式: {"action": "动作", "confidence": 0.0-1.0, "params": {}, "reasoning": "推理过程", "needs_confirm": true/false}"""


def disambiguate(text: str, nlp_result: Optional[dict], api_key: str,
                 timeout: float = 5.0) -> Optional[DisambiguateResult]:
    """调用DeepSeek API进行消歧"""
    if not api_key: return None
    try:
        user_msg = f"用户输入: {text}"
        if nlp_result:
            user_msg += f"\nNLP初步结果: {json.dumps(nlp_result, ensure_ascii=False)}"
        resp = requests.post(
            "https://api.deepseek.com/v1/chat/completions",
            headers={"Authorization": f"Bearer {api_key}", "Content-Type": "application/json"},
            json={"model": "deepseek-chat", "messages": [
                {"role": "system", "content": SYSTEM_PROMPT},
                {"role": "user", "content": user_msg}],
                "temperature": 0.1, "max_tokens": 500},
            timeout=timeout)
        if resp.status_code != 200: return None
        content = resp.json()["choices"][0]["message"]["content"]
        # 提取JSON
        start = content.find("{"); end = content.rfind("}") + 1
        if start < 0 or end <= start: return None
        data = json.loads(content[start:end])
        return DisambiguateResult(
            action=data.get("action", ""),
            confidence=data.get("confidence", 0.0),
            params=data.get("params", {}),
            reasoning=data.get("reasoning", ""),
            needs_confirm=data.get("needs_confirm", True))
    except Exception as e:
        print(f"[LLM] disambiguate error: {e}")
        return None
```

### 5.2 风险分级（risk_classify.py）

```python
"""风险分级 - V1.1增加5级语义分层的影响"""
from dataclasses import dataclass
from enum import Enum
from typing import Optional


class RiskLevel(Enum):
    L1 = "L1"; L2 = "L2"; L3 = "L3"


@dataclass
class RiskResult:
    level: RiskLevel; factors: list[str]; requires_reachability: bool
    requires_flow_precheck: bool; description: str; semantic_level: int = 3


# 动作基础风险等级
ACTION_RISK: dict[str, RiskLevel] = {
    "点动": RiskLevel.L1, "关节运动": RiskLevel.L1, "虚拟轴运动": RiskLevel.L1,
    "回零": RiskLevel.L1, "停止": RiskLevel.L1,
    "暂停": RiskLevel.L1, "继续": RiskLevel.L1, "查询": RiskLevel.L1,
    "复位": RiskLevel.L1, "示教": RiskLevel.L1, "标定": RiskLevel.L1,
    "移动": RiskLevel.L2, "急停": RiskLevel.L2, "打开": RiskLevel.L2, "关闭": RiskLevel.L2,
    "执行": RiskLevel.L3,
}

# 语义层级对风险的影响: L5应急→跳过安全检查，L4系统管理→固定L2
SEMANTIC_LEVEL_RISK_OVERRIDE: dict[int, Optional[RiskLevel]] = {
    5: None,   # L5应急: 跳过风险分级，走独立通道
    4: RiskLevel.L2,  # L4系统管理: 固定L2
}

# 速度阈值
SPEED_THRESHOLD_L2 = 80  # >80% L1→L2
SPEED_THRESHOLD_L3 = 95  # >95% L2→L3
# 距离阈值
DISTANCE_THRESHOLD_L2 = 500  # >500mm L1→L2


def classify_risk(action: str, params: dict, semantic_level: int = 3) -> RiskResult:
    """风险分级: 基础等级 + 速度/距离因子提升 + 语义层级影响"""
    # L5应急: 跳过风险分级
    if semantic_level == 5:
        return RiskResult(RiskLevel.L1, [], False, False, "应急指令-跳过风险分级", 5)

    base = ACTION_RISK.get(action, RiskLevel.L2)
    level = base; factors = []

    # L4系统管理: 固定L2
    if semantic_level == 4:
        level = RiskLevel.L2
        factors.append("系统管理指令")

    speed = params.get("speed_pct", 50)
    distance = params.get("distance_mm", 0)
    # 速度因子
    if speed > SPEED_THRESHOLD_L3:
        if level == RiskLevel.L1: level = RiskLevel.L3; factors.append(f"速度>{SPEED_THRESHOLD_L3}%")
        elif level == RiskLevel.L2: level = RiskLevel.L3; factors.append(f"速度>{SPEED_THRESHOLD_L3}%")
    elif speed > SPEED_THRESHOLD_L2:
        if level == RiskLevel.L1: level = RiskLevel.L2; factors.append(f"速度>{SPEED_THRESHOLD_L2}%")
    # 距离因子
    if distance > DISTANCE_THRESHOLD_L2 and level == RiskLevel.L1:
        level = RiskLevel.L2; factors.append(f"距离>{DISTANCE_THRESHOLD_L2}mm")
    # 需要额外检查
    needs_reach = (action == "移动" and level.value >= "L2")
    needs_flow = (action == "执行")
    desc = f"基础{base.value}"
    if factors: desc += f" + {', '.join(factors)} -> {level.value}"
    return RiskResult(level, factors, needs_reach, needs_flow, desc, semantic_level)
```

### 5.3 确认模式判断（confirm_mode.py）

```python
"""确认模式判断 - V1.1增加新手/熟练/专家三级"""
from dataclasses import dataclass
from enum import Enum
from typing import Optional


class ConfirmMode(Enum):
    NONE = "none"; VOICE = "voice"; DUAL = "dual"  # 无需/语音/双重(语音+屏幕)


class OperatorLevel(Enum):
    NOVICE = "novice"      # 新手: 所有操作需确认
    SKILLED = "skilled"    # 熟练: L1免确认，L2语音确认

@dataclass
class ConfirmResult:
    needs_confirm: bool; mode: ConfirmMode; timeout_seconds: float; message: str


# 免确认动作 (安全指令随时执行)
SKIP_CONFIRM_ACTIONS = {"急停", "停止", "暂停", "继续", "查询", "解除急停"}
# 风险等级→确认模式映射（按操作者级别）
RISK_CONFIRM_MAP = {
    # (risk_level, operator_level) -> (needs_confirm, mode)
    ("L1", OperatorLevel.NOVICE): (True, ConfirmMode.VOICE),
    ("L1", OperatorLevel.SKILLED): (False, ConfirmMode.NONE),
    ("L1", OperatorLevel.EXPERT): (False, ConfirmMode.NONE),
    ("L2", OperatorLevel.NOVICE): (True, ConfirmMode.DUAL),
    ("L2", OperatorLevel.SKILLED): (True, ConfirmMode.VOICE),
    ("L2", OperatorLevel.EXPERT): (False, ConfirmMode.NONE),
    ("L3", OperatorLevel.NOVICE): (True, ConfirmMode.DUAL),
    ("L3", OperatorLevel.SKILLED): (True, ConfirmMode.DUAL),
    ("L3", OperatorLevel.EXPERT): (True, ConfirmMode.VOICE),
}

# 超时时间（秒）
TIMEOUT_MAP = {
    ConfirmMode.VOICE: 10.0,
    ConfirmMode.DUAL: 30.0,
}


def check_confirm_needed(action: str, risk_level: str, params: dict,
                         operator_level: OperatorLevel = OperatorLevel.SKILLED) -> ConfirmResult:
    """判断是否需要确认及确认模式 - 支持三级操作者"""
    if action in SKIP_CONFIRM_ACTIONS:
        return ConfirmResult(False, ConfirmMode.NONE, 0, f"{action}免确认")

    key = (risk_level, operator_level)
    needs, mode = RISK_CONFIRM_MAP.get(key, (True, ConfirmMode.VOICE))

    if not needs:
        return ConfirmResult(False, mode, 0, f"{action}低风险无需确认(级别:{operator_level.value})")

    timeout = TIMEOUT_MAP.get(mode, 10.0)
    level_desc = {"novice": "新手", "skilled": "熟练", "expert": "专家"}
    msg = f"{action}需要{'语音' if mode == ConfirmMode.VOICE else '双重'}确认(级别:{level_desc.get(operator_level.value, '')})"
    return ConfirmResult(True, mode, timeout, msg)
```

### 5.4 L1安全预检（l1_precheck.py）

```python
"""L1安全预检 - V1.1增加球面半径检查和速度钳位检查"""
import math
from dataclasses import dataclass, field
from typing import Optional


@dataclass
class L1CheckItem:
    name: str; passed: bool; message: str


@dataclass
class L1PrecheckResult:
    all_passed: bool; checks: list[L1CheckItem]; summary: str


# 软限位范围 (mm)
SOFT_LIMITS = {
    "target_x": (-800.0, 800.0), "target_y": (-800.0, 800.0), "target_z": (-200.0, 1000.0),
}
# 使能轴列表 (默认6轴全使能)
DEFAULT_ENABLED_AXES = [0, 1, 2, 3, 4, 5]
# 球面半径限制 (mm) - 机械臂最大伸展半径
MAX_SPHERE_RADIUS = 1200.0
# 速度钳位 - 不同动作的最大允许速度百分比
SPEED_CLAMPS: dict[str, int] = {
    "关节运动": 50, "虚拟轴运动": 80, "移动": 100, "回零": 50, "标定": 30,
}


def l1_safety_precheck(action: str, params: dict, system_state: dict) -> L1PrecheckResult:
    """L1安全预检: 软限位/速度/轴使能/急停状态/球面半径/速度钳位"""
    checks = []
    # 1. 急停状态检查 (急停指令允许在急停状态下执行)
    estop = system_state.get("emergency_stop", False)
    if estop and action != "急停":
        checks.append(L1CheckItem("急停状态", False, "急停已激活，只有急停指令可执行"))
    elif estop and action == "急停":
        checks.append(L1CheckItem("急停状态", True, "急停指令允许在急停状态执行"))
    else:
        checks.append(L1CheckItem("急停状态", True, "急停未激活"))
    # 2. 软限位检查
    limit_ok = True; limit_msg = ""
    for key, (lo, hi) in SOFT_LIMITS.items():
        val = params.get(key)
        if val is not None and (val < lo or val > hi):
            limit_ok = False; limit_msg += f"{key}={val}超出[{lo},{hi}]; "
    checks.append(L1CheckItem("软限位", limit_ok, limit_msg if not limit_ok else "目标位置在安全范围内"))
    # 3. 速度合理性检查
    speed = params.get("speed_pct", 50)
    speed_ok = 1 <= speed <= 100
    checks.append(L1CheckItem("速度合理性", speed_ok, f"速度{speed}%超出范围[1,100]" if not speed_ok else f"速度{speed}%合理"))
    # 4. 轴使能检查
    axis_id = params.get("axis_id") or params.get("axis_no")
    enabled = system_state.get("enabled_axes", DEFAULT_ENABLED_AXES)
    if axis_id is not None:
        axis_ok = axis_id in enabled
        checks.append(L1CheckItem("轴使能", axis_ok, f"轴{axis_id}未使能" if not axis_ok else f"轴{axis_id}已使能"))
    else:
        checks.append(L1CheckItem("轴使能", True, "无需检查特定轴"))
    # 5. 球面半径检查（对齐需求V2.0）
    tx = params.get("target_x", 0.0); ty = params.get("target_y", 0.0); tz = params.get("target_z", 0.0)
    if action == "移动" and (tx != 0 or ty != 0 or tz != 0):
        radius = math.sqrt(tx**2 + ty**2 + tz**2)
        radius_ok = radius <= MAX_SPHERE_RADIUS
        checks.append(L1CheckItem("球面半径", radius_ok,
            f"目标半径{radius:.1f}mm超过最大{MAX_SPHERE_RADIUS}mm" if not radius_ok else f"目标半径{radius:.1f}mm在安全范围内"))
    else:
        checks.append(L1CheckItem("球面半径", True, "无需检查"))
```

### 6.2 MODBUS写入（modbus_writer.py）

```python
"""MODBUS写入 - V1.1 7步握手协议，严格对齐通讯协议V4.3"""
import time
from dataclasses import dataclass
from typing import Optional, Callable


@dataclass
class ModbusWriteResult:
    success: bool; func_id: int; message: str; write_time_ms: float


# 关键地址定义（修正！）
IEEE_FUNC_ID = 0       # 函数号
IEEE_TRIGGER = 32      # 触发位（修正：V1.0错误地用1）
LONG_EXEC_STATE = 34   # 执行状态
LONG_ALARM_CODE = 38   # 报警码
IEEE_ECHO_START = 280  # 回显区起始地址

# 急停函数不需要检查状态
EMERGENCY_FUNC_IDS = {104}


class ModbusWriter:
    def __init__(self):
        self._write_func = None; self._read_func = None

    def set_io_functions(self, write_func: Callable, read_func: Callable):
        self._write_func = write_func; self._read_func = read_func

    def modbus_write(self, func_id: int, ieee_dict: dict[int, float],
                     timeout_ms: float = 5000.0) -> ModbusWriteResult:
        """7步握手协议 - 严格按通讯协议V4.3"""
        t0 = time.time()
        if not self._write_func or not self._read_func:
            return ModbusWriteResult(False, func_id, "IO未设置", 0)
        try:
            # 步骤1: 写入函数号
            self._write_func(IEEE_FUNC_ID, func_id)
            # 步骤2: 写入参数（触发前）
            for addr, val in sorted(ieee_dict.items()):
                if addr not in (IEEE_FUNC_ID, IEEE_TRIGGER):
                    self._write_func(addr, val)
            # 步骤3: 读回显校验（关键！V1.0缺失）
            if func_id not in EMERGENCY_FUNC_IDS:
                echo_ok = self._verify_echo(func_id, ieee_dict)
                if not echo_ok:
                    return ModbusWriteResult(False, func_id, "回显校验失败", (time.time()-t0)*1000)
            # 步骤4: 检查状态标志（急停函数跳过）
            if func_id not in EMERGENCY_FUNC_IDS:
                state_ok = self._check_state(func_id)
                if not state_ok:
                    return ModbusWriteResult(False, func_id, "状态检查未通过", (time.time()-t0)*1000)
            # 步骤5: 触发执行
            self._write_func(IEEE_TRIGGER, 1)  # 修正：IEEE(32)=1
            # 步骤6: 等待完成
            deadline = time.time() + timeout_ms / 1000.0
            while time.time() < deadline:
                state = self._read_func(LONG_EXEC_STATE)
                # 根据func_id检查对应的2位状态
                func_state = self._get_func_state(func_id, state)
                if func_state == 2:  # 完成
                    break
                if func_state == 3:  # 错误
                    alarm = self._read_func(LONG_ALARM_CODE)
                    return ModbusWriteResult(False, func_id, f"执行错误,报警码:{alarm}", (time.time()-t0)*1000)
                time.sleep(0.05)
            else:
                return ModbusWriteResult(False, func_id, "超时", (time.time()-t0)*1000)
            # 步骤7: 检查最终状态
            alarm = self._read_func(LONG_ALARM_CODE)
            if alarm != 0:
                return ModbusWriteResult(False, func_id, f"报警码:{alarm}", (time.time()-t0)*1000)
            return ModbusWriteResult(True, func_id, "完成", (time.time()-t0)*1000)
        except Exception as e:
            return ModbusWriteResult(False, func_id, str(e), (time.time()-t0)*1000)

    def _verify_echo(self, func_id: int, ieee_dict: dict) -> bool:
        """步骤3: 回显校验 - 读IEEE(280~312)确认数据正确写入"""
        try:
            echo_func = self._read_func(IEEE_ECHO_START)  # IEEE(280)=函数号
            if int(echo_func) != func_id:
                return False
            # 可选：校验关键参数回显
            return True
        except:
            return True  # 读取失败时降级通过

    def _check_state(self, func_id: int) -> bool:
        """步骤4: 检查LONG(34)状态标志"""
        try:
            state = self._read_func(LONG_EXEC_STATE)
            # 检查: 无报警(bit24=0) + 未急停(bit25=0) + 系统就绪(bit28=1) + 对应通道空闲
            if state & 0x1000000: return False  # 报警
            if state & 0x2000000: return False  # 急停
            if not (state & 0x10000000): return False  # 未就绪
            # 检查对应函数通道空闲
            func_state = self._get_func_state(func_id, state)
            if func_state != 0: return False  # 非空闲
            return True
        except:
            return True

    def _get_func_state(self, func_id: int, long34: int) -> int:
        """从LONG(34)提取对应函数的2位状态"""
        MASK_MAP = {104: 0x3, 106: 0xC, 107: 0x30, 108: 0xC0,
                    109: 0x300, 110: 0xC00, 11: 0xC000, 120: 0xC0000}
        SHIFT_MAP = {104: 0, 106: 2, 107: 4, 108: 6,
                     109: 8, 110: 10, 11: 14, 120: 18}
        mask = MASK_MAP.get(func_id, 0)
        shift = SHIFT_MAP.get(func_id, 0)
        return (long34 & mask) >> shift

    def emergency_stop(self) -> ModbusWriteResult:
        """急停 - 独立通道，不检查状态"""
        if not self._write_func: return ModbusWriteResult(False, 104, "未设置", 0)
        t0 = time.time()
        try:
            self._write_func(IEEE_FUNC_ID, 104)
            self._write_func(2, 1)  # 急停按下
            self._write_func(IEEE_TRIGGER, 1)
            return ModbusWriteResult(True, 104, "急停已发送", (time.time()-t0)*1000)
        except Exception as e:
            return ModbusWriteResult(False, 104, str(e), (time.time()-t0)*1000)
```

### 6.3 状态轮询（status_poller.py）

```python
"""状态轮询 - V1.1修正状态读取方式（使用LONG(34)位操作）"""
import time
from dataclasses import dataclass
from typing import Optional, Callable


@dataclass
class PollResult:
    success: bool; final_state: int; elapsed_ms: float; error_flag: int; message: str


# LONG(34)位掩码和偏移（对齐6.2节ModbusWriter）
FUNC_MASK_MAP = {104: 0x3, 106: 0xC, 107: 0x30, 108: 0xC0,
                 109: 0x300, 110: 0xC00, 11: 0xC000, 120: 0xC0000}
FUNC_SHIFT_MAP = {104: 0, 106: 2, 107: 4, 108: 6,
                  109: 8, 110: 10, 11: 14, 120: 18}
# 状态值: 0=空闲, 1=执行中, 2=完成, 3=错误
STATE_IDLE = 0; STATE_EXECUTING = 1; STATE_COMPLETED = 2; STATE_ERROR = 3


class StatusPoller:
    def __init__(self):
        self._read_func = None

    def set_read_function(self, read_func: Callable): self._read_func = read_func

    def _get_func_state(self, func_id: int, long34: int) -> int:
        """从LONG(34)提取对应函数的2位状态"""
        mask = FUNC_MASK_MAP.get(func_id, 0)
        shift = FUNC_SHIFT_MAP.get(func_id, 0)
        return (long34 & mask) >> shift

    def poll_execution(self, func_id: int = 0, timeout_s: float = 10.0) -> PollResult:
        """轮询执行状态 - 使用LONG(34)位操作"""
        if not self._read_func: return PollResult(False, -1, 0, 0, "未设置")
        t0 = time.time()
        while time.time() < t0 + timeout_s:
            try:
                long34 = self._read_func(34)
                ef = self._read_func(38)  # 报警码
                if func_id:
                    st = self._get_func_state(func_id, int(long34))
                else:
                    # 无指定func_id时，检查全局状态
                    st = 0 if long34 == 0 else 1
                if st == STATE_COMPLETED:
                    return PollResult(True, st, (time.time()-t0)*1000, ef, "完成")
                if st == STATE_ERROR:
                    return PollResult(False, st, (time.time()-t0)*1000, ef, f"错误: 报警码{ef}")
                if st == STATE_IDLE:
                    return PollResult(True, st, (time.time()-t0)*1000, ef, "空闲")
            except Exception as e:
                return PollResult(False, -1, (time.time()-t0)*1000, 0, str(e))
            time.sleep(0.05)
        return PollResult(False, STATE_EXECUTING, (time.time()-t0)*1000, 0, "超时")

    def wait_for_idle(self, func_id: int = 0, timeout_s: float = 10.0) -> PollResult:
        """等待指定函数通道空闲"""
        if not self._read_func: return PollResult(False, -1, 0, 0, "未设置")
        t0 = time.time()
        while time.time() < t0 + timeout_s:
            try:
                long34 = self._read_func(34)
                if func_id:
                    st = self._get_func_state(func_id, int(long34))
                else:
                    st = 0 if long34 == 0 else 1
                if st == STATE_IDLE:
                    return PollResult(True, 0, (time.time()-t0)*1000, 0, "空闲")
            except: pass
            time.sleep(0.05)
        return PollResult(False, -1, (time.time()-t0)*1000, 0, "超时")
```

### 6.5 并行执行器（parallel_executor.py）

```python
"""并行执行器 - Func110/120可与Func106/107/108并行执行
对齐通讯协议V4.3 1.3节：110(T1延时)和120(IO输出)可与运动函数同时执行，
但110和120之间串行执行(同一任务)，不能同时运行。
"""
import time
import threading
from dataclasses import dataclass, field
from typing import Optional, Callable
from enum import Enum


class ChannelType(Enum):
    MOTION = "motion"    # T0通道: Func106/107/108/109/11
    PARALLEL = "parallel" # T1通道: Func110/120


@dataclass
class ParallelTask:
    task_id: str; func_id: int; ieee_dict: dict[int, float]
    channel: ChannelType; description: str = ""
    result: Optional[dict] = None; completed: bool = False


class ParallelExecutor:
    """并行执行器 - 管理运动通道(T0)和并行通道(T1)的并行执行"""

    def __init__(self):
        self._write_func = None; self._read_func = None
        self._motion_busy = False   # T0通道是否忙碌
        self._parallel_busy = False # T1通道是否忙碌
        self._lock = threading.Lock()
        self._parallel_thread: Optional[threading.Thread] = None

    def set_io_functions(self, write_func: Callable, read_func: Callable):
        self._write_func = write_func; self._read_func = read_func

    def can_start_parallel(self) -> bool:
        """检查是否可以启动并行任务(T1通道空闲)"""
        return not self._parallel_busy

    def execute_parallel(self, task: ParallelTask, callback: Optional[Callable] = None):
        """启动并行任务(T1通道)，不阻塞T0通道的运动指令"""
        if self._parallel_busy:
            if callback: callback({"success": False, "message": "T1通道忙碌，110和120不能同时执行"})
            return
        self._parallel_busy = True
        self._parallel_thread = threading.Thread(
            target=self._run_parallel_task, args=(task, callback), daemon=True)
        self._parallel_thread.start()

    def _run_parallel_task(self, task: ParallelTask, callback: Optional[Callable]):
        """在独立线程中执行并行任务"""
        try:
            # 写入Func参数
            for addr, val in sorted(task.ieee_dict.items()):
                self._write_func(addr, val)
            # 触发
            self._write_func(32, 1)  # IEEE(32)=1 触发
            # 等待完成 - 检查LONG(34)对应位
            mask_map = {110: 0xC00, 120: 0xC0000}
            shift_map = {110: 10, 120: 18}
            mask = mask_map.get(task.func_id, 0)
            shift = shift_map.get(task.func_id, 0)
            deadline = time.time() + 30.0  # 30秒超时
            while time.time() < deadline:
                state = self._read_func(34)  # LONG(34)
                func_state = (state & mask) >> shift if mask else 0
                if func_state == 2:  # 完成
                    task.completed = True
                    task.result = {"success": True, "message": f"{task.description}完成"}
                    break
                if func_state == 3:  # 错误
                    alarm = self._read_func(38)  # LONG(38)
                    task.result = {"success": False, "message": f"执行错误,报警码:{alarm}"}
                    break
                time.sleep(0.05)
            else:
                task.result = {"success": False, "message": "超时"}
        except Exception as e:
            task.result = {"success": False, "message": str(e)}
        finally:
            self._parallel_busy = False
            if callback: callback(task.result)

    # 典型并行场景：运动到位后延时再闭合夹爪
    # (1) 触发108运动到目标位置（T0通道）
    # (2) 同时触发110延时0.5秒（T1通道并行）
    # (3) 等110完成(DONE)
    # (4) 触发120闭合夹爪（T1通道串行，等110完成后执行）
    # (5) 等120完成(DONE)
    # (6) 触发下一个运动指令（T0通道）
```

```python
"""确认处理 - V1.1增加三级确认模式（新手/熟练/专家）"""
import time
from dataclasses import dataclass
from typing import Optional, Callable
from enum import Enum


class ConfirmAction(Enum):
    CONFIRMED = "confirmed"; CANCELLED = "cancelled"; TIMEOUT = "timeout"


class OperatorLevel(Enum):
    NOVICE = "novice"      # 新手
    SKILLED = "skilled"    # 熟练
    EXPERT = "expert"      # 专家


@dataclass
class ConfirmState:
    pending_action: str; pending_params: dict; confirm_message: str
    timeout_seconds: float; created_at: float; mode: str = "voice"
    operator_level: str = "skilled"


class ConfirmHandler:
    CONFIRMED_WORDS = ["确认", "是", "好的", "执行", "要得", "可以", "对", "对头"]
    CANCELLED_WORDS = ["取消", "不", "不要", "算了", "要不得", "不对头"]

    def __init__(self, operator_level: OperatorLevel = OperatorLevel.SKILLED):
        self._pending = None; self._on_confirmed = None; self._on_cancelled = None
        self._operator_level = operator_level

    def set_operator_level(self, level: OperatorLevel):
        """设置操作者级别"""
        self._operator_level = level

    def set_callbacks(self, on_confirmed: Callable, on_cancelled: Callable):
        self._on_confirmed = on_confirmed; self._on_cancelled = on_cancelled

    def request_confirm(self, action: str, params: dict, message: str,
                        timeout: float = 10.0, mode: str = "voice") -> bool:
        if self._pending: return False
        self._pending = ConfirmState(action, params, message, timeout, time.time(), mode,
                                     self._operator_level.value)
        return True

    def handle_response(self, response: str) -> Optional[ConfirmAction]:
        if not self._pending: return None
        if response in self.CONFIRMED_WORDS:
            s = self._pending; self._pending = None
            if self._on_confirmed: self._on_confirmed(s.pending_action, s.pending_params)
            return ConfirmAction.CONFIRMED
        elif response in self.CANCELLED_WORDS:
            s = self._pending; self._pending = None
            if self._on_cancelled: self._on_cancelled(s.pending_action, s.pending_params)
            return ConfirmAction.CANCELLED
        return None

    def check_timeout(self) -> Optional[ConfirmAction]:
        if not self._pending: return None
        if time.time() - self._pending.created_at > self._pending.timeout_seconds:
            s = self._pending; self._pending = None
            if self._on_cancelled: self._on_cancelled(s.pending_action, s.pending_params)
            return ConfirmAction.TIMEOUT
        return None

    @property
    def is_pending(self) -> bool: return self._pending is not None
    @property
    def pending_message(self) -> Optional[str]: return self._pending.confirm_message if self._pending else None
    @property
    def operator_level(self) -> OperatorLevel: return self._operator_level
```

```python
        total = progress.get("total", self._total_steps); axis = progress.get("axis", 0)
        direction = progress.get("direction", ""); remaining = progress.get("remaining", "")

        # 根据进度百分比选择模板
        if pct >= 90 and len(templates) > 2:
            msg = templates[2]

            msg = templates[1]
```

```python
import json, os
from dataclasses import dataclass, asdict
from typing import Optional
from datetime import datetime


@dataclass
class PositionEntry:
    name: str; description: str = ""
    x: float = 0.0; y: float = 0.0; z: float = 0.0
    rx: float = 0.0; ry: float = 0.0; rz: float = 0.0
    spd: int = 50                     # 默认速度百分比
    move_type: int = 0                # 运动类型: 0=关节运动, 1=直线插补
    locked: bool = False              # 锁定标志: True=锁定不可修改/删除
    created_at: str = ""; updated_at: str = ""; is_system: bool = False

    def to_func104_params(self, speed_pct: int = 0) -> dict:
        """转换为Func108直线插补参数（move_type=1）或Func106关节运动参数（move_type=0）"""
        spd = speed_pct if speed_pct > 0 else self.spd
        return {"motion_mode": self.move_type,
                "target_x": self.x, "target_y": self.y, "target_z": self.z,
                "target_rx": self.rx, "target_ry": self.ry, "target_rz": self.rz,
                "speed_pct": spd, "position_name": self.name}


class PositionRegistry:
    def __init__(self, filepath: str = "position_registry.json"):
        self.filepath = filepath
        self._positions: dict[str, PositionEntry] = {}
        self._permission = None  # 权限管理器引用，由外部注入
        self._load()

    def set_permission_manager(self, pm):
        """注入权限管理器"""
        self._permission = pm

    def _check_permission(self, action: str) -> bool:
        """检查权限，无权限管理器时默认允许"""
        if self._permission is None:
            return True
        return self._permission.check_permission(action)

    def _load(self):
        if os.path.exists(self.filepath):
            with open(self.filepath, "r", encoding="utf-8") as f:
                for item in json.load(f).get("positions", []):
                    e = PositionEntry(**item)
                    self._positions[e.name] = e

    def _save(self):
        data = {"version": "1.1", "updated_at": datetime.now().isoformat(),
                "positions": [asdict(e) for e in self._positions.values()]}
        with open(self.filepath, "w", encoding="utf-8") as f:
            json.dump(data, f, ensure_ascii=False, indent=2)

    def add(self, entry: PositionEntry) -> tuple[bool, str]:
        """添加位置，需要工程师权限"""
        if not self._check_permission("保存位置"):
            return False, "此操作需要工程师权限"
        if entry.name in self._positions:
            return False, f"位置'{entry.name}'已存在"
        entry.created_at = datetime.now().isoformat()
        entry.updated_at = entry.created_at
        self._positions[entry.name] = entry
        self._save()
        return True, f"位置'{entry.name}'已保存"

    def remove(self, name: str) -> tuple[bool, str]:
        """删除位置，需要工程师权限，锁定位置不可删除"""
        if not self._check_permission("删除位置"):
            return False, "此操作需要工程师权限"
        e = self._positions.get(name)
        if e is None:
            return False, f"位置'{name}'不存在"
        if e.is_system or e.locked:
            return False, f"位置'{name}'已锁定，不可删除"
        del self._positions[name]
        self._save()
        return True, f"位置'{name}'已删除"

    def update(self, name: str, **kwargs) -> tuple[bool, str]:
        """修改位置，需要工程师权限，锁定位置不可修改"""
        if not self._check_permission("修改位置"):
            return False, "此操作需要工程师权限"
        e = self._positions.get(name)
        if e is None:
            return False, f"位置'{name}'不存在"
        if e.locked and not kwargs.get("unlock", False):
            return False, f"位置'{name}'已锁定，不可修改"
        for k, v in kwargs.items():
            if hasattr(e, k) and k not in ("name", "is_system", "created_at"):
                setattr(e, k, v)
        e.updated_at = datetime.now().isoformat()
        self._save()
        return True, f"位置'{name}'已更新"

    def set_position(self, name: str, **kwargs) -> tuple[bool, str]:
        """设定位置坐标（示教记录），需要工程师权限"""
        if not self._check_permission("设定位置"):
            return False, "此操作需要工程师权限"
        return self.update(name, **kwargs)

    def get(self, name: str) -> Optional[PositionEntry]:
        return self._positions.get(name)

    def list_all(self) -> list[PositionEntry]:
        return list(self._positions.values())

    def search(self, keyword: str) -> list[PositionEntry]:
        kw = keyword.lower()
        return [e for e in self._positions.values()
                if kw in e.name.lower() or kw in e.description.lower()]
```

```python
        FlowState.IDLE: {FlowState.READY, FlowState.REHEARSAL},
        FlowState.READY: {FlowState.RUNNING, FlowState.IDLE, FlowState.REHEARSAL},
        FlowState.RUNNING: {FlowState.PAUSED, FlowState.COMPLETED, FlowState.ERROR},
        FlowState.PAUSED: {FlowState.RUNNING, FlowState.IDLE},
        FlowState.COMPLETED: {FlowState.IDLE},
        FlowState.ERROR: {FlowState.IDLE},
        FlowState.REHEARSAL: {FlowState.IDLE, FlowState.READY},
    }

    def __init__(self, filepath: str = "flow_registry.json"):
        self.filepath = filepath
        self._flows: dict[str, FlowEntry] = {}
        self._permission = None
        self._load()

    def set_permission_manager(self, pm):
        """注入权限管理器"""
        self._permission = pm

    def _check_permission(self, action: str) -> bool:
        if self._permission is None:
            return True
        return self._permission.check_permission(action)

    def _load(self):
        if os.path.exists(self.filepath):
            with open(self.filepath, "r", encoding="utf-8") as f:
                for item in json.load(f).get("flows", []):
                    steps = [FlowStep(**s) for s in item.get("steps", [])]
                    e = FlowEntry(**{k: v for k, v in item.items() if k != "steps"})
                    e.steps = steps
                    self._flows[e.name] = e

    def _save(self):
        data = {"version": "1.1", "updated_at": datetime.now().isoformat(),
                "flows": [asdict(f) for f in self._flows.values()]}
        with open(self.filepath, "w", encoding="utf-8") as f:
            json.dump(data, f, ensure_ascii=False, indent=2)

    def add(self, entry: FlowEntry) -> tuple[bool, str]:
        """新建流程，需要工程师权限"""
        if not self._check_permission("新建流程"):
            return False, "此操作需要工程师权限"
        if entry.name in self._flows:
            return False, f"流程'{entry.name}'已存在"
        entry.created_at = datetime.now().isoformat()
        entry.updated_at = entry.created_at
        self._flows[entry.name] = entry
        self._save()
        return True, f"流程'{entry.name}'已创建"

    def remove(self, name: str) -> tuple[bool, str]:
        """删除流程，需要工程师权限"""
        if not self._check_permission("删除流程"):
            return False, "此操作需要工程师权限"
        if name not in self._flows:
            return False, f"流程'{name}'不存在"
        del self._flows[name]
        self._save()
        return True, f"流程'{name}'已删除"

    def update(self, name: str, **kwargs) -> tuple[bool, str]:
        """修改流程，需要工程师权限"""
        if not self._check_permission("修改流程"):
            return False, "此操作需要工程师权限"
        f = self._flows.get(name)
        if f is None:
            return False, f"流程'{name}'不存在"
        for k, v in kwargs.items():
            if hasattr(f, k) and k not in ("name", "created_at"):
                setattr(f, k, v)
        f.updated_at = datetime.now().isoformat()
        self._save()
        return True, f"流程'{name}'已更新"

    def confirm(self, name: str) -> tuple[bool, str]:
        """确认流程，需要工程师权限"""
        if not self._check_permission("确认流程"):
            return False, "此操作需要工程师权限"
        f = self._flows.get(name)
        if f is None:
            return False, f"流程'{name}'不存在"
        f.confirmed = True
        f.updated_at = datetime.now().isoformat()
        self._save()
        return True, f"流程'{name}'已确认"

    def start_rehearsal(self, name: str) -> tuple[bool, str]:
        """启动演练模式，需要工程师权限，使用演练速度"""
        if not self._check_permission("演练流程"):
            return False, "此操作需要工程师权限"
        f = self._flows.get(name)
        if f is None:
            return False, f"流程'{name}'不存在"
        if not self.transition(name, FlowState.REHEARSAL):
            return False, f"流程'{name}'当前状态不允许演练"
        return True, f"流程'{name}'演练模式启动，速度{f.rehearsal_spd}%"

    def get(self, name: str) -> Optional[FlowEntry]:
        return self._flows.get(name)

    def list_all(self) -> list[FlowEntry]:
        return list(self._flows.values())

    def transition(self, name: str, target: FlowState) -> bool:
        f = self._flows.get(name)
        if f is None:
            return False
        cur = FlowState(f.state)
        if target not in self.VALID_TRANSITIONS.get(cur, set()):
            return False
        f.state = target.value
        f.updated_at = datetime.now().isoformat()
        self._save()
        return True

    def advance_step(self, name: str) -> Optional[FlowStep]:
        f = self._flows.get(name)
        if f is None:
            return None
        f.current_step += 1
        if f.current_step > len(f.steps):
            f.current_step = 0
            self.transition(name, FlowState.COMPLETED)
            return None
        self._save()
        return f.steps[f.current_step - 1]

    def reset(self, name: str) -> bool:
        f = self._flows.get(name)
        if f is None:
            return False
        f.current_step = 0
        f.state = FlowState.IDLE.value
        self._save()
        return True
```

```python
"""报警检测 - 50ms周期，对齐通讯协议V4.3 LONG(38)位定义"""
import time
from dataclasses import dataclass
from typing import Optional, Callable


@dataclass
class AlarmInfo:
    code: int; message: str; axis: int; timestamp: float
    level: str; auto_clear: bool


# 报警码定义 - 对齐通讯协议V4.3 LONG(38)位定义
# 标志位  掩码    含义            置1条件             level    auto_clear
ALARM_CODES = {
    0x01: {"msg": "半径超限",        "level": "warning", "auto_clear": True},   # 38.00 $1
    0x02: {"msg": "高度超限",        "level": "warning", "auto_clear": True},   # 38.01 $2
    0x04: {"msg": "指令忙(重复触发)", "level": "warning", "auto_clear": True},   # 38.02 $4
    0x08: {"msg": "速度超限",        "level": "warning", "auto_clear": True},   # 38.03 $8
    0x10: {"msg": "加速度超限",      "level": "error",   "auto_clear": False},  # 38.04 $10
    0x20: {"msg": "减速度超限",      "level": "error",   "auto_clear": False},  # 38.05 $20
    0x40: {"msg": "ECAT通讯故障",    "level": "error",   "auto_clear": True},   # 38.06 $40
    0x80: {"msg": "驱动器报警",      "level": "error",   "auto_clear": True},   # 38.07 $80
    0x100:{"msg": "函数号非法",      "level": "error",   "auto_clear": False},  # 38.08 $100
    0x200:{"msg": "参数非法",        "level": "error",   "auto_clear": False},  # 38.09 $200
}


class AlarmDetector:
    def __init__(self):
        self._read_func = None
        self._alarm_cb = None
        self._last_code = 0

    def set_read_function(self, read_func: Callable):
        self._read_func = read_func

    def set_alarm_callback(self, cb: Callable[[AlarmInfo], None]):
        self._alarm_cb = cb

    def check(self) -> list[AlarmInfo]:
        """检查报警 - LONG(38)为位掩码，需逐位解析"""
        if not self._read_func:
            return []
        try:
            raw_code = int(self._read_func(38))  # LONG(38)
            if raw_code == 0:
                self._last_code = 0
                return []
            # 逐位解析报警码
            alarms = []
            for bit_mask, alarm_def in ALARM_CODES.items():
                if raw_code & bit_mask:
                    # 查找报警轴
                    axis = -1
                    for i in range(6):
                        try:
                            if self._read_func(274 + i):
                                axis = i
                                break
                        except Exception:
                            pass
                    info = AlarmInfo(
                        code=bit_mask,
                        message=alarm_def["msg"],
                        axis=axis,
                        timestamp=time.time(),
                        level=alarm_def["level"],
                        auto_clear=alarm_def["auto_clear"]
                    )
                    alarms.append(info)
                    if self._alarm_cb:
                        self._alarm_cb(info)
            self._last_code = raw_code
            return alarms
        except Exception:
            return []

    def check_single(self) -> Optional[AlarmInfo]:
        """兼容旧接口：返回最高优先级报警"""
        alarms = self.check()
        if not alarms:
            return None
        # error级别优先于warning
        for a in alarms:
            if a.level == "error":
                return a
        return alarms[0]
```

```python
import time
from dataclasses import dataclass
from typing import Optional, Callable


@dataclass
class PipelineResult:

ACK_TEMPLATES = {

    "text": "系统在线，正在处理",

class EmergencyChannel:
    """应急独立通道 - 跳过所有安全检查，直接写入MODBUS"""

    def __init__(self):
        self._write_func = None
        self._whitelist = ["123", "456", "789"]  # 应急编码白名单

    def set_write_function(self, write_func: Callable):
        self._write_func = write_func

    def set_whitelist(self, whitelist: list[str]):
        self._whitelist = whitelist

    def handle_emergency(self, prefix: str, code: Optional[str] = None) -> dict:
        """处理应急指令"""
        t0 = time.time()
        executed = False
        message = ""

        if prefix == "急停":
            # 无条件急停
            if self._write_func:
                self._write_func(0, 104)   # 函数号
                self._write_func(2, 1)     # 急停按下
                self._write_func(32, 1)    # 触发
                executed = True
                message = "急停已执行！"
            else:
                message = "急停写入失败：通讯未连接"

        elif code and code in self._whitelist:
            # 编码验证通过，执行急停
            if self._write_func:
                self._write_func(0, 104)
                self._write_func(2, 1)
                self._write_func(32, 1)
                executed = True
                message = f"应急编码{code}验证通过，急停已执行！"
            else:
                message = "急停写入失败：通讯未连接"

        elif code and code not in self._whitelist:
            message = f"应急编码{code}无效，急停未执行"

        else:
            # 暂停/继续等
            if self._write_func:
                self._write_func(0, 104)
                if prefix == "暂停":
                    self._write_func(4, 1)  # 暂停按下
                elif prefix == "继续":
                    self._write_func(4, 2)  # 暂停松开
                self._write_func(32, 1)
                executed = True
                message = f"{prefix}已执行"
            else:
                message = f"{prefix}写入失败：通讯未连接"

        return {"executed": executed, "message": message, "elapsed_ms": (time.time()-t0)*1000}


class Pipeline:
    def __init__(self, config: dict):
        self.config = config
        # 初始化各模块（导入见1.2模块划分）
        from .nlp.prefix_check import classify_input
        from .nlp.conflict_check import check_command_conflict
        from .nlp.pinyin_normalize import pinyin_normalize
        from .nlp.homophone_map import apply_homophone_map
        from .nlp.dialect_sichuan import dialect_normalize
        from .nlp.unit_normalize import normalize_unit
        from .nlp.element_extract import extract_elements
        from .nlp.llm_disambiguate import disambiguate
        from .safety.default_values import fill_defaults
        from .safety.risk_classify import classify_risk
        from .safety.confirm_mode import check_confirm_needed
        from .safety.l1_precheck import l1_safety_precheck
        from .safety.l2_reachability import check_reachability
        from .safety.l3_flow_precheck import l3_flow_precheck
        from .execution.func_mapper import map_to_func
        from .execution.modbus_writer import ModbusWriter
        from .execution.status_poller import StatusPoller
        from .execution.feedback import generate_feedback
        from .response.reply_template import get_reply
        from .response.confirm_handler import ConfirmHandler
        from .data.position_registry import PositionRegistry
        from .data.flow_registry import FlowRegistry
        from .data.dialog_logger import DialogLogger
        from .data.memory_params import MemoryManager
        from .data.permission_manager import PermissionManager
        # 存储引用
        self._classify = classify_input
        self._conflict = check_command_conflict
        self._pinyin = pinyin_normalize
        self._homophone = apply_homophone_map
        self._dialect = dialect_normalize
        self._unit = normalize_unit
        self._extract = extract_elements
        self._llm = disambiguate
        self._defaults = fill_defaults
        self._risk = classify_risk
        self._confirm = check_confirm_needed
        self._l1 = l1_safety_precheck
        self._l2 = check_reachability
        self._l3 = l3_flow_precheck
        self._map = map_to_func
        self._feedback = generate_feedback
        self._reply = get_reply
        # 有状态模块
        self.modbus = ModbusWriter()
        self.poller = StatusPoller()
        self.confirm_handler = ConfirmHandler()
        self.positions = PositionRegistry(config.get("position_file", "position_registry.json"))
        self.flows = FlowRegistry(config.get("flow_file", "flow_registry.json"))
        self.logger = DialogLogger(config.get("log_dir", "logs"))
        self.memory = MemoryManager(config.get("memory_file", "memory_params.json"))
        self.permission = PermissionManager(config.get("emergency", {}).get("engineer_password", "720111"))
        self.emergency_channel = EmergencyChannel()
        # 注入权限管理器
        self.positions.set_permission_manager(self.permission)
        self.flows.set_permission_manager(self.permission)
        # 回执应答回调
        self._ack_cb: Optional[Callable] = None

    def set_ack_callback(self, cb: Callable):
        """设置回执应答回调函数"""
        self._ack_cb = cb

    def _extract_action(self, text: str) -> str:
        """从文本中提取动作关键词"""
        action_keywords = ["保存位置", "删除位置", "设定位置", "修改位置",
                          "新建流程", "修改流程", "删除流程", "确认流程", "演练流程"]
        for kw in action_keywords:
            if kw in text:
                return kw
        return ""

    def process(self, text: str) -> PipelineResult:
        t0 = time.time(); details = {}

        # === 应急独立通道（最高优先级，跳过语义分析）===
        cr = self._classify(text)
        if cr.input_type == InputType.EMERGENCY:
            # 立即发送回执
            ack_msg = ACK_TEMPLATES["emergency"]
            if self._ack_cb: self._ack_cb(ack_msg)
            # 走应急通道
            er = self.emergency_channel.handle_emergency(
                cr.matched_prefix or "急停", cr.emergency_code)
            reply = er["message"]
            self.logger.log_input(text)
            self.logger.log_output(reply, "emergency", "ok" if er["executed"] else "err")
            return PipelineResult(er["executed"], "emergency", reply, er["elapsed_ms"], details)

        # === 非应急指令：发送回执应答 ===
        if self._ack_cb:
            ack_msg = ACK_TEMPLATES.get("text", "系统在线，正在处理")
            self._ack_cb(ack_msg)

        # L1 闲聊：直接回复
        if cr.semantic_level == 1:
            self.logger.log_input(text)
            msg = "你好！我是机械手语音助手小正。说'小正'加指令即可操作。"
            self.logger.log_output(msg)
            return PipelineResult(True, "chitchat", msg, (time.time()-t0)*1000, details)

        # L2 查询/工艺研讨
        if cr.input_type == InputType.QUERY:
            result = self._handle_query(cr.cleaned_text)
            self.logger.log_input(text); self.logger.log_output(result)
            return PipelineResult(True, "query", result, (time.time()-t0)*1000, details)

        if cr.input_type == InputType.PRECHECK:
            result = self._handle_precheck(cr.cleaned_text)
            self.logger.log_input(text); self.logger.log_output(result)
            return PipelineResult(True, "precheck", result, (time.time()-t0)*1000, details)

        # L4 系统管理：权限检查
        if cr.input_type == InputType.SYSTEM:
            action = self._extract_action(cr.cleaned_text)
            if not self.permission.check_permission(action):
                msg = f"此操作需要工程师权限，请切换到工程师界面"
                self.logger.log_input(text); self.logger.log_output(msg)
                return PipelineResult(False, "permission", msg, (time.time()-t0)*1000, details)

        # L3 常规生产执行：完整流水线
        # 1a.冲突检查
        state = self._get_state()
        cf = self._conflict(cr.cleaned_text, state)
        if cf.has_conflict:
            msg = cf.message + " " + cf.suggestion
            self.logger.log_input(text); self.logger.log_output(msg, result="conflict")
            return PipelineResult(False, "conflict", msg, (time.time()-t0)*1000, details)

        # 2-5. NLP处理链（拼音→同音字→方言→单位→要素提取）
        norm, _ = self._pinyin(cr.cleaned_text)
        norm = self._homophone(norm); norm, _ = self._dialect(norm)
        units = self._unit(norm)
        ext = self._extract(norm)
        params = {k: getattr(ext, k) for k in ["action","target_position","direction","distance_mm","angle_deg","speed_pct","axis_id","io_number"]}
        for uv in units:
            if uv.unit == "mm" and params.get("distance_mm") is None: params["distance_mm"] = uv.value
            elif uv.unit == "deg" and params.get("angle_deg") is None: params["angle_deg"] = uv.value
            elif uv.unit == "%" and params.get("speed_pct") is None: params["speed_pct"] = int(uv.value)
        action = ext.action
        if action is None:
            lr = self._llm(text, params, self.config.get("deepseek_api_key", ""))
            if lr: action = lr.action; params.update(lr.params)
            else: return PipelineResult(False, "unknown", "无法识别", (time.time()-t0)*1000, details)

        # 6.默认值补全（使用记忆参数）
        pd = None
        if params.get("target_position"):
            pe = self.positions.get(params["target_position"])
            if pe: pd = {"x": pe.x, "y": pe.y, "z": pe.z, "rx": pe.rx, "ry": pe.ry, "rz": pe.rz, "spd": pe.spd, "move_type": pe.move_type}
        params = self._defaults(action, params, self.memory, pd)

        # 7.风险分级
        risk = self._risk(action, params); details["risk"] = risk.level.value

        # 8.确认模式
        cnf = self._confirm(action, risk.level.value, params)
        if cnf.needs_confirm:
            msg = self._reply(action, "confirm", **params)
            self.confirm_handler.request_confirm(action, params, msg, cnf.timeout_seconds, cnf.mode.value)
            self.logger.log_input(text); self.logger.log_output(msg, action, "waiting")
            return PipelineResult(True, action, msg, (time.time()-t0)*1000, details)

        # 11.L1预检
        l1 = self._l1(action, params, state)
        if not l1.all_passed:
            self.logger.log_input(text); self.logger.log_output(l1.summary, action, "l1_fail")
            return PipelineResult(False, action, l1.summary, (time.time()-t0)*1000, details)

        # L2可达性
        if risk.requires_reachability and action == "移动":
            r = self._l2(params.get("target_x",0), params.get("target_y",0), params.get("target_z",0),
                         params.get("target_rx",0), params.get("target_ry",0), params.get("target_rz",0),
                         self.config.get("zmotion_handle"))
            if not r.reachable:
                self.logger.log_input(text); self.logger.log_output(r.message, action, "l2_fail")
                return PipelineResult(False, action, r.message, (time.time()-t0)*1000, details)

        # 12.Func映射
        fm = self._map(action, params)
        if fm is None: return PipelineResult(False, action, "映射失败", (time.time()-t0)*1000, details)

        # 13.MODBUS写入（7步握手+回显校验）
        wr = self.modbus.modbus_write(fm["func_id"], fm["ieee_dict"], self.config.get("modbus_timeout", 5000))

        # 14.状态轮询
        pr = self.poller.poll_execution(self.config.get("poll_timeout", 10.0))

        # 15.结果反馈
        ok = wr.success and pr.success
        fb = self._feedback(action, ok, params, wr.message or pr.message, (time.time()-t0)*1000)
        mode = "normal" if fb.level == "success" else "error"
        reply = self._reply(action, mode, **params, reason=fb.text)
        self.logger.log_input(text); self.logger.log_output(reply, action, "ok" if ok else "err")
        # 更新记忆参数
        if ok:
            self.memory.update_after_command(action, params)
        return PipelineResult(ok, action, reply or fb.text, (time.time()-t0)*1000, details)

    def _handle_query(self, text: str) -> str:
        """L2查询：直接读取看板缓存，≤500ms返回"""
        summary = self.dashboard.get_summary()
        if "位置" in text or "坐标" in text:
            p = summary["position"]
            return f"当前位置: X={p['x']:.1f} Y={p['y']:.1f} Z={p['z']:.1f}"
        if "状态" in text:
            return f"设备状态: {summary['system_state']}, 报警: {'有' if summary['alarm_active'] else '无'}, 急停: {'是' if summary['estop'] else '否'}"
        if "速度" in text:
            return f"当前速度: {summary['speed_pct']:.0f}%"
        if "通讯" in text:
            return f"通讯状态: {'正常' if summary['ecat_ok'] else '异常'}"
        return f"设备状态: {summary['system_state']}"

    def _handle_precheck(self, text: str) -> str:
        """L2工艺研讨：预检查询"""
        # 提取目标坐标进行L1+L2预检
        return "预检功能待实现（需要L1+L2预检）"

    def _get_state(self) -> dict:
        return {"is_moving": False, "alarm_active": False, "is_homed": True,
                "emergency_stop": False, "flow_running": False, "is_paused": False}
```

### 10.2 主循环

```python
"""主循环 - PySide6事件循环+语音回调+50ms看板刷新+50ms报警检测+回执应答"""
import sys
from PySide6.QtWidgets import QApplication
from PySide6.QtCore import QTimer


class MainApplication:
    def __init__(self):
        self.app = QApplication(sys.argv)
        self.pipeline = Pipeline(self._load_config())
        self.input_handler = InputHandler()
        self.input_handler.set_process_callback(self.pipeline.process)
        self.dashboard = DashboardCache()
        self.alarm_detector = AlarmDetector()
        # 设置回执应答回调
        self.pipeline.set_ack_callback(self._on_ack)
        # 50ms定时器
        self._timer = QTimer()
        self._timer.timeout.connect(self._tick)
        self._timer.start(50)

    def _on_ack(self, ack_msg: str):
        """回执应答回调 - 语音播报回执"""
        self.input_handler.speak(ack_msg)

    def _tick(self):
        """50ms周期任务"""
        self.dashboard.refresh()       # 看板刷新
        self.alarm_detector.check()    # 报警检测
        self.pipeline.confirm_handler.check_timeout()  # 确认超时

    def _load_config(self) -> dict:
        import yaml
        try:
            with open("config.yaml", "r", encoding="utf-8") as f:
                return yaml.safe_load(f)
        except: return {}

    def run(self):
        # 初始化语音
        cfg = self.pipeline.config
        self.input_handler.initialize_voice(
            cfg.get("iflytek_app_id", ""),
            cfg.get("iflytek_api_key", ""),
            cfg.get("iflytek_api_secret", "")
        )
        self.input_handler.start_voice()
        sys.exit(self.app.exec())


if __name__ == "__main__":
    MainApplication().run()
```
