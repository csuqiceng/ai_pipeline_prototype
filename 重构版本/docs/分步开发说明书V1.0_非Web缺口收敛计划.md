# 分步开发说明书 V1.0 非 Web 缺口收敛 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 将《机械手自然语言交互系统_分步开发说明书_V1.0.md》中的非 Web 需求与当前 V5.0 Qt 实现收敛，补齐真实缺口，同时明确不再照搬的 V4.3/greenfield 设计。

**Architecture:** 不重建 `robot_voice/` 目录，不重写 Qt/mixin 主架构。新增少量 facade 和配置层来兼容 V1.0 文档要求，核心控制链路继续保持 V5.0 三道门、Func104 可审计应急通道、现有 `RobotModbusService`/`VoiceNlpAdapter`/`SafetyPrecheckService`。

**Tech Stack:** Python 3.10+、PySide6、pytest、JSON 配置、现有 `robot_modbus_lite` 包。

| 类型 | 处理方式 |

| V1.0 要求 | 当前决策 | 原因 |
|---|---|---|
| V4.3 七步握手 | 不实现 | 当前主线为 V5.0 三道门：回显比对、触发前复查、控制器接受确认 |

```python
from __future__ import annotations

from dataclasses import dataclass, field


Pose6 = tuple[float, float, float, float, float, float]


@dataclass
class MotionHistory:
    position_stack: list[Pose6] = field(default_factory=list)
    last_direction: dict[str, float | int] | None = None
    last_command: dict[str, str | int] | None = None

    def push_position(self, pose: Pose6) -> None:
        values = tuple(float(value) for value in pose)
        if len(values) != 6:
            raise ValueError("pose must contain exactly 6 values")
        self.position_stack.append(values)  # type: ignore[arg-type]

    def pop_position(self) -> Pose6 | None:
        if not self.position_stack:
            return None
        return self.position_stack.pop()

    def record_direction(self, *, axis_no: int, direction: int, step: float) -> None:
        self.last_direction = {
            "axis_no": int(axis_no),
            "direction": int(direction),
            "step": float(step),
        }

    def get_last_direction(self) -> dict[str, float | int] | None:
        return None if self.last_direction is None else dict(self.last_direction)

    def record_command(self, text: str, *, func_id: int) -> None:
        self.last_command = {"text": str(text), "func_id": int(func_id)}

    def get_last_command(self) -> dict[str, str | int] | None:
        return None if self.last_command is None else dict(self.last_command)
```

def _operator_try_modify_pending_confirm(self, text: str) -> bool:
    import re

    plan = getattr(self, "_operator_pending_confirm_plan", None)
    if plan is None:
        return False
    match = re.search(r"(?:速度|速率|spd)(?:改成|设为|设置为|到)?\s*(\d+(?:\.\d+)?)", str(text or ""))
    if not match:
        return False
    speed = max(5.0, min(150.0, float(match.group(1))))
    records = getattr(plan, "atomic_records", {}) or {}
    for record in records.values():
        params = getattr(record, "params", None)
        if isinstance(params, dict):
            params["spd_pct"] = speed
            params["acc_pct"] = speed
            params["dec_pct"] = speed
    return True

```python
from __future__ import annotations

import json
from pathlib import Path
from typing import Any


class ConfigManager:
    def __init__(self, config_dir: str | Path):
        self.config_dir = Path(config_dir)
        self._cache: dict[str, dict[str, Any]] = {}

    def load_config(self, config_name: str) -> dict[str, Any]:
        key = str(config_name).removesuffix(".json")
        path = self.config_dir / f"{key}.json"
        if not path.exists():
            raise FileNotFoundError(path)
        data = json.loads(path.read_text(encoding="utf-8"))
        if not isinstance(data, dict):
            raise ValueError(f"{path} must contain a JSON object")
        self._cache[key] = data
        return dict(data)

    def get(self, key: str, default: Any = None) -> Any:
        parts = str(key).split(".")
        if not parts:
            return default
        config_name = parts[0]
        data = self._cache.get(config_name)
        if data is None:
            try:
                data = self.load_config(config_name)
            except Exception:
                return default
        current: Any = data
        for part in parts[1:]:
            if not isinstance(current, dict) or part not in current:
                return default
            current = current[part]
        return current

    def reload(self) -> None:
        names = list(self._cache)
        self._cache.clear()
        for name in names:
            self.load_config(name)
```

```python
from __future__ import annotations

from datetime import datetime
from typing import Any
from uuid import uuid4


class JsonMessage:
    @staticmethod
    def build_command(func_id: int, ieee_dict: dict[int | str, Any], metadata: dict[str, Any] | None = None) -> dict[str, Any]:
        return {
            "msg_type": "command",
            "msg_id": uuid4().hex,
            "timestamp": datetime.now().isoformat(),
            "func_id": int(func_id),
            "ieee": {str(key): value for key, value in ieee_dict.items()},
            "metadata": dict(metadata or {}),
        }

    @staticmethod
    def build_query(query_type: str, params: dict[str, Any] | None = None) -> dict[str, Any]:
        return {
            "msg_type": "query",
            "msg_id": uuid4().hex,
            "timestamp": datetime.now().isoformat(),
            "query_type": str(query_type),
            "params": dict(params or {}),
        }

    @staticmethod
    def build_feedback(
        func_id: int,
        success: bool,
        detail: str,
        device_status: dict[str, Any] | None = None,
        safety_result: dict[str, Any] | None = None,
        emergency: dict[str, Any] | None = None,
    ) -> dict[str, Any]:
        return {
            "msg_type": "feedback",
            "msg_id": uuid4().hex,
            "timestamp": datetime.now().isoformat(),
            "func_id": int(func_id),
            "success": bool(success),
            "detail": str(detail),
            "device_status": dict(device_status or {}),
            "safety_result": dict(safety_result or {}),
            "emergency": dict(emergency or {}),
        }

    @staticmethod
    def parse_message(data: dict[str, Any]) -> tuple[bool, str]:
        if not isinstance(data, dict):
            return False, "message must be dict"
        if not data.get("msg_type"):
            return False, "missing msg_type"
        if not data.get("msg_id"):
            return False, "missing msg_id"
        return True, ""
```
